% \umlicon{x,y}: UML component icon whose top-right corner is at (x,y).
\newcommand{\umlicon}[1]{%
  \draw[fill=white] ($(#1)+(-2.6,0)$) rectangle +(2.6,-2.0);
  \draw[fill=white] ($(#1)+(-3.1,-0.4)$) rectangle +(1.0,-0.45);
  \draw[fill=white] ($(#1)+(-3.1,-1.2)$) rectangle +(1.0,-0.45);
}
% \umlcomponent{name}{x}{y}{width in mm}{height in mm}{text}: component box with icon.
\newcommand{\umlcomponent}[6]{%
  \node[diagram box=#5mm,text width=#4mm-2mm] (#1) at (#2,#3) {#6};
  \umlicon{#1.north east}
}
% \umlassembly{x1,y1}{x2,y2}: ball-and-socket connector; the first point requires
% the interface and the second provides it.
\newcommand{\umlassembly}[2]{%
  \coordinate (assembly mid) at ($(#1)!0.5!(#2)$);
  \draw (#1) -- ($(assembly mid)!1.5mm!(#1)$);
  \draw (#2) -- (assembly mid);
  \draw[fill=white] (assembly mid) circle (0.9mm);
  \draw let \p1=($(#1)-(#2)$), \n1={atan2(\y1,\x1)} in
    ($(assembly mid)+({\n1-90}:1.5mm)$) arc[start angle=\n1-90,end angle=\n1+90,radius=1.5mm];
}
% \umlnode[style]{x1}{y1}{x2}{y2}{stereotype}{name}: UML deployment node (cube);
% (x1,y1) and (x2,y2) are the corners of the front face.
\newcommand{\umlnode}[7][]{%
  \draw[fill=gray!25,#1] (#2,#5) -- ($(#2,#5)+(2.5,2.5)$) -- ($(#4,#5)+(2.5,2.5)$)
    -- ($(#4,#3)+(2.5,2.5)$) -- (#4,#3) -- (#4,#5) -- cycle;
  \draw[fill=gray!12,#1] (#2,#3) rectangle (#4,#5);
  \node[align=center,anchor=north,text width={(#4-#2)*1mm-2mm}] at ({(#2+#4)/2},#5-0.8)
    {\diagramstereotype{#6}\\{\bfseries #7}};
}
% Sequence-diagram helpers. Coordinates are in mm, with y running down from 0 at the
% lifeline heads. Draw order in a figure: lifelines, fragments, bars, then messages.
\tikzset{
  seq call/.style={-{Stealth[length=1.7mm,width=1.3mm]}},
  seq async/.style={-{Straight Barb[length=1.7mm,width=1.5mm]}},
  seq return/.style={dashed,-{Straight Barb[length=1.7mm,width=1.5mm]}},
}
% \seqlifeline{x}{bottom y}{name}
\newcommand{\seqlifeline}[3]{%
  \draw[dashed,gray] (#1,-3.5) -- (#1,#2);
  \node[diagram box=11mm,text width=15mm] at (#1,2) {#3};
}
% \seqbar{x}{top y}{bottom y}: activation bar
\newcommand{\seqbar}[3]{\draw[fill=white] ({#1-0.8},#2) rectangle ({#1+0.8},#3);}
% \seqmsg{style}{from x}{to x}{y}{label}: message between activation bars
\newcommand{\seqmsg}[5]{%
  \pgfmathsetmacro{\seqdir}{ifthenelse(#3>#2,1,-1)}
  \draw[#1] ({#2+\seqdir*0.8},#4) -- node[above,flow label,fill=white,inner sep=0.2mm,yshift=0.2mm] {#5} ({#3-\seqdir*0.8},#4);
}
% \seqself{x}{y}{label} and \seqselfleft{x}{y}{label}: self-message
\newcommand{\seqself}[3]{%
  \draw[seq call] ({#1+0.8},#2) -- ++(4,0) -- ++(0,-2.5) -- ++(-4,0);
  \node[anchor=west,flow label,fill=white] at ({#1+5.2},{#2-1.25}) {#3};
}
\newcommand{\seqselfleft}[3]{%
  \draw[seq call] ({#1-0.8},#2) -- ++(-4,0) -- ++(0,-2.5) -- ++(4,0);
  \node[anchor=east,flow label,fill=white] at ({#1-5.2},{#2-1.25}) {#3};
}
% \seqfrag{x1}{top y}{x2}{bottom y}{keyword}{guard x}{guard}: combined fragment
\newcommand{\seqfrag}[7]{%
  \draw (#1,#2) rectangle (#3,#4);
  \node[anchor=north west,draw,fill=white,inner sep=0.6mm,font=\diagramfont\bfseries] at (#1,#2) {#5};
  \node[anchor=north west,flow label,fill=white] at (#6,{#2-0.3}) {[#7]};
}
% \seqelse{x1}{x2}{y}{guard x}{guard}: dashed separator between alt operands
\newcommand{\seqelse}[5]{%
  \draw[dashed] (#1,#3) -- (#2,#3);
  \node[anchor=north west,flow label,fill=white] at (#4,{#3-0.3}) {[#5]};
}

\chapter*{Template Instructions}
\addcontentsline{toc}{chapter}{Template Instructions}
This template supports development, research, hybrid, and game projects. Complete
all common sections and retain conditional sections only when they apply to the
selected project. Keep the document concise and use cross-references instead of
repeating information.

\begin{itemize}
  \item Replace every \placeholder{placeholder} and remove all example text before submission.
  \item State the project type on the title page: Development, Research, Hybrid, or Game.
  \item Assign stable identifiers to objectives, scope items, requirements, risks, and design elements.
  \item Write one clear idea or testable requirement per numbered item.
  \item Use standard UML notation where UML diagrams are included.
  \item Cite every external idea, dataset, standard, figure, and factual claim.
  \item Keep detailed test cases and results in the separate software testing document; reference them through the traceability matrix.
\end{itemize}

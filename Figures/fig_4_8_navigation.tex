% \screens{prefix}{x}{list of level/label}: level 0 = main screen, level 1 = opened from the screen above it
\newcommand{\screens}[3]{%
  \foreach \lvl/\t [count=\i] in {#3}{%
    \pgfmathsetmacro\off{.42*\lvl}
    \pgfmathsetmacro\bw{4.1-.84*\lvl}\pgfmathsetmacro\tw{4.1-.84*\lvl-.2}
    \ifnum\lvl=0
      \node[scr] (#1\i) at (#2,{-3.55-.66*\i}) {\t};
    \else
      \node[sub,minimum width=\bw cm,text width=\tw cm] (#1\i) at ({#2+\off},{-3.55-.66*\i}) {\t};
      \draw[semithick] ({#2-2.05+.42*\lvl},{-3.55-.66*\i+.42}) |- (#1\i.west);
    \fi}}
